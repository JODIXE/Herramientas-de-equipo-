    <script>
        function updateNomenclature() {
            // Esta lógica es de ejemplo imitando la imagen,
            // asumiendo que solo se actualiza la REF al final.
            const referencia = document.getElementById('referencia').value || 'XXX';
            
            // Texto base imitando la imagen de ejemplo
            const baseText = "ACREDITACION POR PAGO NO PROCESADO PROVEEDOR: BDV REF: ";
            
            document.getElementById('resultText').innerText = baseText + referencia;
        }

        function copyNomenclature() {
            const textToCopy = document.getElementById('resultText').innerText;
            navigator.clipboard.writeText(textToCopy).then(() => {
                // Opcional: una alerta sutil
                // alert('Copiado al portapapeles');
            });
        }

        function clearFields() {
            // Restablece los selects al primer elemento
            document.getElementById('metodo').selectedIndex = 0;
            document.getElementById('proveedor').selectedIndex = 0;
            document.getElementById('variante').selectedIndex = 0;
            // Limpia la referencia
            document.getElementById('referencia').value = '';
            // Actualiza el resultado
            updateNomenclature();
        }

        // Inicializar al cargar
        updateNomenclature();
    </script>
